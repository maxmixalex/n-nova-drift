cmake_minimum_required(VERSION 3.16)
project(n_nova_drift LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NOVA_ENABLE_SDL "Build a desktop preview window when SDL2 is installed" ON)
option(NOVA_SANITIZE "Enable host address/undefined behavior sanitizers" OFF)
include(CTest)
find_package(Python3 COMPONENTS Interpreter QUIET)

# Propagate the same diagnostics and sanitizer runtime to every host target.
add_library(nova-build-options INTERFACE)
if(MSVC)
  target_compile_options(nova-build-options INTERFACE /W4 /utf-8)
else()
  target_compile_options(nova-build-options INTERFACE -Wall -Wextra)
endif()
if(NOVA_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(nova-build-options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(nova-build-options INTERFACE -fsanitize=address,undefined)
endif()

# Compile the large game core once; all regression programs exercise the same
# object files as the desktop application instead of recompiling each fixture.
set(NOVA_CORE_SOURCES src/game.cpp src/content.cpp src/source_waves.cpp)
add_library(nova-core STATIC ${NOVA_CORE_SOURCES})
target_include_directories(nova-core PUBLIC src include)
target_link_libraries(nova-core PUBLIC nova-build-options)
file(GLOB NOVA_SOURCES CONFIGURE_DEPENDS src/*.cpp)
set(NOVA_UI_SOURCES ${NOVA_SOURCES})
list(FILTER NOVA_UI_SOURCES EXCLUDE REGEX "/(main|game|content|source_waves)\\.cpp$")
add_executable(nova-drift src/main.cpp ${NOVA_UI_SOURCES})
target_link_libraries(nova-drift PRIVATE nova-core)
if(NOVA_ENABLE_SDL)
  find_package(SDL2 QUIET)
  if(SDL2_FOUND)
    target_compile_definitions(nova-drift PRIVATE NOVA_SDL=1)
    if(TARGET SDL2::SDL2)
      target_link_libraries(nova-drift PRIVATE SDL2::SDL2)
    else()
      target_include_directories(nova-drift PRIVATE ${SDL2_INCLUDE_DIRS})
      target_link_libraries(nova-drift PRIVATE ${SDL2_LIBRARIES})
    endif()
  else()
    message(STATUS "SDL2 development files absent: host preview supports --headless only")
  endif()
endif()
add_executable(nova-converter tools/converter/main.cpp)
target_include_directories(nova-converter PRIVATE include)
target_link_libraries(nova-converter PRIVATE nova-build-options)
if(NOT MSVC AND NOT NOVA_SANITIZE)
  target_compile_options(nova-converter PRIVATE -O3)
endif()

if(BUILD_TESTING)
  function(nova_test target test_name)
    add_executable(${target} ${ARGN})
    target_include_directories(${target} PRIVATE src include)
    target_link_libraries(${target} PRIVATE nova-build-options)
    # Test setup also uses assert: keep those expressions in Release builds.
    if(MSVC)
      target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${target} PRIVATE -Wpedantic -UNDEBUG)
    endif()
    if(NOT "${test_name}" STREQUAL "")
      add_test(NAME ${test_name} COMMAND ${target})
    endif()
  endfunction()
  function(nova_core_test target test_name source)
    nova_test(${target} ${test_name} ${source})
    target_link_libraries(${target} PRIVATE nova-core)
  endfunction()
