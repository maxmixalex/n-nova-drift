cmake_minimum_required(VERSION 3.16)
project(n_nova_drift LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NOVA_ENABLE_SDL "Build a desktop preview window when SDL2 is installed" ON)
option(NOVA_SANITIZE "Enable host address/undefined behavior sanitizers" OFF)
file(GLOB NOVA_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_executable(nova-drift ${NOVA_SOURCES})
target_include_directories(nova-drift PRIVATE src include)
target_compile_options(nova-drift PRIVATE -Wall -Wextra)
if(NOVA_ENABLE_SDL)
  find_package(SDL2 QUIET)
  if(SDL2_FOUND)
    target_compile_definitions(nova-drift PRIVATE NOVA_SDL=1)
    if(TARGET SDL2::SDL2)
      target_link_libraries(nova-drift PRIVATE SDL2::SDL2)
    else()
      target_include_directories(nova-drift PRIVATE ${SDL2_INCLUDE_DIRS})
      target_link_libraries(nova-drift PRIVATE ${SDL2_LIBRARIES})
    endif()
  else()
    message(STATUS "SDL2 development files absent: host preview supports --headless only")
  endif()
endif()
if(NOVA_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(nova-drift PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(nova-drift PRIVATE -fsanitize=address,undefined)
endif()
find_package(Python3 COMPONENTS Interpreter QUIET)
add_executable(nova-converter tools/converter/main.cpp)
target_include_directories(nova-converter PRIVATE include)
target_compile_options(nova-converter PRIVATE -O3 -Wall -Wextra)
include(CTest)
if(BUILD_TESTING)
  add_executable(nova-game-tests tests/game_test.cpp src/game.cpp src/content.cpp)
  target_include_directories(nova-game-tests PRIVATE src)
  # The regression executable uses assert; Release must still execute checks
  # and the setup operations within those expressions.
  target_compile_options(nova-game-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  if(NOVA_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(nova-game-tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(nova-game-tests PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME gameplay COMMAND nova-game-tests)
  add_test(NAME converter-codec COMMAND nova-converter --self-test)
  set(NOVA_APP_TEST_SOURCES ${NOVA_SOURCES})
  list(FILTER NOVA_APP_TEST_SOURCES EXCLUDE REGEX "/main\\.cpp$")
  add_executable(app-flow-tests tests/app_flow.cpp ${NOVA_APP_TEST_SOURCES})
  target_include_directories(app-flow-tests PRIVATE src include)
  target_compile_definitions(app-flow-tests PRIVATE NOVA_TEST=1)
  target_compile_options(app-flow-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  if(NOVA_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(app-flow-tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(app-flow-tests PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME app-flow COMMAND app-flow-tests)
endif()
if(Python3_Interpreter_FOUND)
  add_custom_target(ndless
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/build_ndless.py
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    COMMENT "Build ARM926 Ndless .tns launcher")
endif()
